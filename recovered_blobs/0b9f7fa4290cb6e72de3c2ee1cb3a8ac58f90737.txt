// 精準的多邊形重疊檢查函式 (Precision Polygon Overlap Check)
const checkPolygonOverlap = (polyA, polyB) => {
    const segmentsIntersect = (p1, p2, p3, p4) => {
        const ccw = (A, B, C) => (C.y - A.y) * (B.x - A.x) > (B.y - A.y) * (C.x - A.x);
        return ccw(p1, p3, p4) !== ccw(p2, p3, p4) && ccw(p1, p2, p3) !== ccw(p1, p2, p4);
    };

    for (let i = 0; i < polyA.length; i++) {
        const a1 = polyA[i];
        const a2 = polyA[(i + 1) % polyA.length];
        for (let j = 0; j < polyB.length; j++) {
            const b1 = polyB[j];
            const b2 = polyB[(j + 1) % polyB.length];
            if (segmentsIntersect(a1, a2, b1, b2)) {
                return true;
            }
        }
    }

    const isPointInPolygon = (pt, poly) => {
        let inside = false;
        for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
            const xi = poly[i].x, yi = poly[i].y;
            const xj = poly[j].x, yj = poly[j].y;
            const intersect = ((yi > pt.y) !== (yj > pt.y))
                && (pt.x < (xj - xi) * (pt.y - yi) / (yj - yi) + xi);
            if (intersect) inside = !inside;
        }
        return inside;
    };

    if (polyA.length > 0 && isPointInPolygon(polyA[0], polyB)) return true;
    if (polyB.length > 0 && isPointInPolygon(polyB[0], polyA)) return true;

    return false;
};

                    const rx = dot >= 0 ? nx : -nx;
                    const ry = dot >= 0 ? ny : -ny;
                    
                    const roadOutP1 = { x: p1.x + rx * roadWidth, y: p1.y + ry * roadWidth };
                    const roadOutP2 = { x: p2.x + rx * roadWidth, y: p2.y + ry * roadWidth };
                    roadWorldPolygon = [p1, p2, roadOutP2, roadOutP1];
                }
            }
        }

        // 精準檢查與所有現有基地是否重疊
        let isColliding = false;
        for (const s of sites) {
            const sPos = s.position || { x: 0, y: 0, rotation: 0 };
            const sGeom = s.baseGeometry?.points || s.basePoints;
            let sLocalPts = [];
            
            const ew = parseFloat(s.inputs?.baseWidthM || s.inputs?.baseWidth) || Math.sqrt(s.inputs?.baseArea || 239);
            const ed = parseFloat(s.inputs?.baseDepthM) || ((s.inputs?.baseArea || 239) / ew);
            
            if (sGeom && sGeom.length > 2) {
                sLocalPts = sGeom;
            } else {
                sLocalPts = [
                    { x: -ew/2, y: -ed/2 },
                    { x: ew/2, y: -ed/2 },
                    { x: ew/2, y: ed/2 },
                    { x: -ew/2, y: ed/2 }
                ];
            }
            
            const cosS = Math.cos(sPos.rotation || 0);
            const sinS = Math.sin(sPos.rotation || 0);
            const worldPtsS = sLocalPts.map(p => ({
                x: sPos.x + (p.x * cosS - p.y * sinS),
                y: sPos.y + (p.x * sinS + p.y * cosS)
            }));
            
            if (checkPolygonOverlap(worldPtsNew, worldPtsS)) {
                isColliding = true;
                break;
            }
        }

        // 檢查與主要道路範圍是否重疊
        if (!isColliding && roadWorldPolygon) {
            if (checkPolygonOverlap(worldPtsNew, roadWorldPolygon)) {
                alert("快速新增位置侵佔了主要面前道路空間，請調整原基地朝向或手動精確放置！");
                return;
            }
        }

        if (isColliding) {
            alert("快速新增位置與現有基地重疊，請調整原基地朝向或手動精確放置！");
            return;
        }

        // 智慧面前道路對齊 (Smart Road Alignment) for 快速新增
        if (siteA && siteA.inputs?.roads) {
            const mainRoad = siteA.inputs.roads.find(r => r.type === 'primary') || siteA.inputs.roads[0];
            if (mainRoad) {
                const roadEdgeP1 = roadWorldPolygon ? roadWorldPolygon[0] : { x: 0, y: 0 };
                const roadEdgeP2 = roadWorldPolygon ? roadWorldPolygon[1] : { x: 0, y: 0 };
                const roadMid = { x: (roadEdgeP1.x + roadEdgeP2.x) / 2, y: (roadEdgeP1.y + roadEdgeP2.y) / 2 };
                
                let minDst = Infinity;
                let bestEdgeIdx = 2;
                
                for (let j = 0; j < 4; j++) {
                    const p1 = worldPtsNew[j];
                    const p2 = worldPtsNew[(j + 1) % 4];
                    const mid = { x: (p1.x + p2.x) / 2, y: (p1.y + p2.y) / 2 };
                    const dst = Math.pow(mid.x - roadMid.x, 2) + Math.pow(mid.y - roadMid.y, 2);
                    if (dst < minDst) {
                        minDst = dst;
                        bestEdgeIdx = j;
                    }
                }
                
                newSite.inputs.roads = [{
                    ...mainRoad,
                    edgeIndex: bestEdgeIdx,
                    edgeIndices: [bestEdgeIdx],
                    orientation: bestEdgeIdx === 0 ? 'bottom' : bestEdgeIdx === 1 ? 'right' : bestEdgeIdx === 2 ? 'top' : 'left'
                }];
            }
        }

        newSite.position = newPos;

        setSites(prev => [...prev, newSite]);
        setActiveSiteIndex(newIdx);
        setShowAddSiteDialog(false);
    };
    const handleRemoveSite = (idx) => {
        if (sites.length <= 1) return;
        if (confirm(`確定要刪除 ${sites[idx].name} 嗎？`)) {
            setSites(prev => prev.filter((_, i) => i !== idx));
            setActiveSiteIndex(prev => prev > idx ? prev - 1 : (prev === idx ? Math.max(0, idx - 1) : prev));
        }
    };
    const handleUpdateSiteGeometry = (idx, geometry) => {
        setSites(prev => {
            const next = [...prev];
            const oldGeom = prev[idx].baseGeometry;
            const newGeom = geometry;

            // Sync adjacent sites if points count remained the same (simple dragging)
            if (oldGeom && oldGeom.points && newGeom && newGeom.points && oldGeom.points.length === newGeom.points.length) {
                const tolerance = 0.5; // 0.5m tolerance for "coincident" points
                const deltas = [];
                for (let i = 0; i < oldGeom.points.length; i++) {
                    const op = oldGeom.points[i];
                    const np = newGeom.points[i];
                    const dx = np.x - op.x;
                    const dy = np.y - op.y;
                    if (Math.abs(dx) > 0.01 || Math.abs(dy) > 0.01) {
                        deltas.push({ op, dx, dy });
                    }
                }

                if (deltas.length > 0) {
                    for (let k = 0; k < next.length; k++) {
                        if (k === idx) continue;
                        let modified = false;
                        const otherGeom = next[k].baseGeometry;
                        if (otherGeom && otherGeom.points) {
                            const newOtherPoints = otherGeom.points.map(p => {
                                let newP = { ...p };
                                for (const d of deltas) {
                                    if (Math.abs(p.x - d.op.x) < tolerance && Math.abs(p.y - d.op.y) < tolerance) {
                                        newP.x += d.dx;
                                        newP.y += d.dy;
                                        modified = true;
                                    }
                                }
                                return newP;
                            });
                            if (modified) {
                                next[k] = { ...next[k], baseGeometry: { ...otherGeom, points: newOtherPoints } };
                            }
                        }
                    }
                }
            }
            
            next[idx] = { ...next[idx], baseGeometry: geometry };
            return next;
        });
    };

    useEffect(() => {
        // Clear totalCFA override when base area or ratio changes
        // eslint-disable-next-line
        setOverrides(prev => { 
            if ('totalCFA' in prev) {
                const copy = { ...prev }; 
                delete copy.totalCFA; 
                return copy; 
            }
            return prev;
        });
    }, [inputs.baseArea, inputs.floorRatio, setOverrides, activeSiteIndex]);

    const handleInputChange = (e) => { const { name, value } = e.target; setInputs(prev => ({ ...prev, [name]: value })); };
    const handleOverride = (id, value) => { const val = parseFloat(value); if (!isNaN(val)) setOverrides(prev => ({ ...prev, [id]: val })); };
    const handleParamChange = (key, value) => { setParams(prev => ({ ...prev, [key]: parseFloat(value) || 0 })); };
    const updateBaseDim = (key, val) => { setInputs(prev => ({ ...prev, [key]: val })); };
    const handleUpdateInputs = (key, val) => { setInputs(prev => ({ ...prev, [key]: val })); };
    const handleUpdateSiteSetting = (id, key, val) => {
        setSites(prev => prev.map(s => s.id === id ? { ...s, [key]: val } : s));
    };

    const inputClass = "w-full p-3 border-2 border-[var(--border-strong)] rounded-none outline-none theme-input text-sm font-bold num-font transition-all duration-200";
    const labelClass = "block text-xs font-black theme-text-muted uppercase mb-1 tracking-wider";

    return (
        <div className="min-h-screen theme-bg px-4 md:px-12 py-8 theme-text selection:bg-[var(--text-accent)] selection:text-white">
            <div className="w-full max-w-none mx-auto space-y-12">
                {/* HEADER */}
                <header className="flex flex-col md:flex-row justify-between items-start md:items-center theme-panel p-8 md:p-12">
                    <div className="space-y-4">
                        <h1 className="text-4xl md:text-6xl font-black flex items-center gap-4 theme-text-title leading-tight">
                            <Building2 className="text-[var(--text-accent)]" size={48} strokeWidth={3} />
                            自地自建<br className="md:hidden" />評估計算書
                        </h1>
                        <p className="theme-text font-bold text-lg opacity-80 border-l-8 border-[var(--text-accent)] pl-4">
                            基地開發效益評估 / 容積獎勵檢討 / 停車空間規劃
                        </p>
                    </div>
                    <div className="flex flex-col md:flex-row items-start md:items-center gap-6 mt-8 md:mt-0">
                        <div className="flex flex-col">
                            <label className="text-xs font-black uppercase tracking-widest text-[var(--text-muted)]">專案名稱</label>
                            <input 
                                type="text" 
                                name="projectName" 
                                value={inputs.projectName} 
                                onChange={handleInputChange} 
                                placeholder="輸入專案名稱" 
                                className="border-b-4 border-[var(--border-strong)] focus:border-[var(--text-accent)] outline-none bg-transparent py-2 text-2xl font-black theme-text placeholder:text-stone-400" 
                            />
                        </div>
                        <button 
                            onClick={() => setIsEditMode(!isEditMode)} 
                            className={`w-16 h-16 flex items-center justify-center transition-all duration-200 border-4 border-[var(--border-strong)] ${isEditMode ? 'theme-btn-active' : 'theme-btn'}`}
                        >
                            {isEditMode ? <Settings2 size={28} strokeWidth={3} /> : <Pencil size={28} strokeWidth={3} />}
                        </button>
                    </div>
                </header>

                {/* ADD SITE DIALOG */}
                {showAddSiteDialog && (
                    <div className="fixed inset-0 z-[100] flex items-center justify-center">
                        <div className="absolute inset-0 bg-black/40 backdrop-blur-sm" onClick={() => setShowAddSiteDialog(false)} />
                        <div className="relative bg-white shadow-2xl border-4 border-black p-8 max-w-md w-full mx-4">
                            <button onClick={() => setShowAddSiteDialog(false)} className="absolute top-4 right-4 text-stone-400 hover:text-black"><X size={20} /></button>
                            <h3 className="text-2xl font-black mb-2 flex items-center gap-3"><MapPinned size={24} className="text-[var(--text-accent)]" /> 新增基地</h3>
                            <p className="text-sm text-stone-500 mb-6">是否緊貼 <span className="font-black text-black">{activeSite.name}</span>？</p>
                            <div className="space-y-3">
                                <button onClick={() => handleAddSite('back')} className="w-full p-4 border-4 border-black font-bold hover:bg-stone-100 text-left flex items-center gap-3 transition-colors">🔗 緊貼後方（背面）</button>
                                <button onClick={() => handleAddSite('left')} className="w-full p-4 border-4 border-black font-bold hover:bg-stone-100 text-left flex items-center gap-3 transition-colors">🔗 緊貼左側</button>
                                <button onClick={() => handleAddSite('right')} className="w-full p-4 border-4 border-black font-bold hover:bg-stone-100 text-left flex items-center gap-3 transition-colors">🔗 緊貼右側</button>
                                <div className="border-t-4 border-stone-200 my-2" />
                                <button onClick={() => handleAddSite('free')} className="w-full p-4 border-4 border-dashed border-stone-400 font-bold hover:bg-stone-50 text-left flex items-center gap-3 text-stone-600 transition-colors">📍 獨立放置（不緊貼）</button>
                            </div>
                        </div>
                    </div>
                )}

                {/* SITE TABS */}
                {sites.length > 0 && (
                    <div className="flex items-center gap-2 flex-wrap">
                        {sites.map((site, idx) => (
                            <div key={site.id} className="relative group">
                                <button
                                    onClick={() => setActiveSiteIndex(idx)}
                                    className={`px-6 py-3 font-black text-lg border-4 border-black transition-all duration-200 flex items-center gap-2 ${
                                        idx === activeSiteIndex
                                            ? 'bg-[var(--text-accent)] text-white shadow-[4px_4px_0_#000] -translate-y-1'
                                            : 'bg-white text-black hover:bg-stone-100'
                                    }`}
                                >
                                    <MapPinned size={18} strokeWidth={3} />
                                    {site.name}
                                </button>
                                {sites.length > 1 && idx === activeSiteIndex && (
                                    <button
                                        onClick={(e) => { e.stopPropagation(); handleRemoveSite(idx); }}
                                        className="absolute -top-2 -right-2 w-6 h-6 bg-red-500 text-white rounded-full flex items-center justify-center text-xs font-black shadow-md hover:bg-red-600 opacity-0 group-hover:opacity-100 transition-opacity"
                                    >
                                        <X size={12} />
                                    </button>
                                )}
                            </div>
                        ))}
                        <button
                            onClick={() => setShowAddSiteDialog(true)}
                            className="px-6 py-3 font-black text-lg border-4 border-dashed border-black text-black hover:bg-stone-100 transition-all duration-200 flex items-center gap-2"
                        >
                            <Plus size={20} strokeWidth={3} /> 新增基地
                        </button>
                    </div>
                )}

                {/* PLACEMENT MODE UI */}
                {placementMode === 'independent' && (
                    <div className="fixed top-20 left-1/2 -translate-x-1/2 z-[100] bg-white border-4 border-black p-6 shadow-[8px_8px_0_var(--text-accent)] flex flex-col gap-4 animate-in fade-in slide-in-from-top-4">
                        <div className="flex items-center justify-between gap-8">
                            <h3 className="text-xl font-black flex items-center gap-2">
                                <MapPinned size={20} className="text-[var(--text-accent)]" /> 精準放置模式
                            </h3>
                            <button onClick={() => { setPlacementMode(false); setReferenceEdge(null); }} className="text-stone-400 hover:text-black transition-colors">
                                <X size={20} />
                            </button>
                        </div>
                        <div className="flex items-center gap-6">
                            <div className="flex flex-col">
                                <span className="text-xs font-bold text-stone-500 uppercase tracking-widest mb-1">參考線 (基準)</span>
                                <span className={`text-sm font-black p-2 border-2 text-center ${referenceEdge ? 'border-emerald-500 text-emerald-700 bg-emerald-50' : 'border-dashed border-stone-300 text-stone-400'}`}>
                                    {referenceEdge ? '已選擇' : '請在 3D 區點擊線段'}
                                </span>
                            </div>
                            <div className="flex flex-col">
                                <span className="text-xs font-bold text-stone-500 uppercase tracking-widest mb-1">X 距離</span>
                                <div className="flex items-center gap-2 border-b-4 border-black pb-1">
                                    <input type="number" 
                                        value={placementOffset.x} onChange={e => setPlacementOffset(p => ({...p, x: e.target.value}))}
                                        className="w-16 text-xl font-black text-center outline-none bg-transparent" placeholder="0" />
                                    <span className="font-bold">m</span>
                                </div>
                            </div>
                            <div className="flex flex-col">
                                <span className="text-xs font-bold text-stone-500 uppercase tracking-widest mb-1">Y 距離</span>
                                <div className="flex items-center gap-2 border-b-4 border-black pb-1">
                                    <input type="number" 
                                        value={placementOffset.y} onChange={e => setPlacementOffset(p => ({...p, y: e.target.value}))}
                                        className="w-16 text-xl font-black text-center outline-none bg-transparent" placeholder="0" />
                                    <span className="font-bold">m</span>
                                </div>
                            </div>
                        </div>
                        <button onClick={handleConfirmPlacement} className="w-full bg-[var(--text-accent)] text-white font-black py-3 hover:bg-opacity-90 active:translate-y-1 transition-all mt-2">
                            確認放置
                        </button>
                    </div>
                )}

                {/* 3D VISUALIZER SECTION */}
                <section className="space-y-4">
                    <div className="theme-panel bg-[var(--text-primary)] p-0">
                        <div 
                            className="p-6 flex justify-between items-center cursor-pointer hover:bg-[var(--text-accent)] transition-colors group" 
                            onClick={() => setShowVisualizer(!showVisualizer)}
                        >
                            <h3 className="text-2xl font-black text-white flex items-center gap-3">
                                <Box size={24} strokeWidth={3} /> 3D 模擬區
                            </h3>
                            <div className="text-white group-hover:scale-125 transition-transform">
                                {showVisualizer ? <ChevronUp size={28} strokeWidth={3} /> : <ChevronDown size={28} strokeWidth={3} />}
                            </div>
                        </div>
                    </div>
                    
                    <div className={`transition-all duration-500 ease-in-out ${showVisualizer ? 'opacity-100' : 'opacity-0 h-0 overflow-hidden'}`}>
                        {showVisualizer && (
                            <div className="theme-panel p-1 bg-[var(--border-strong)]">
                                <Visualizer3D
                                    isDarkMode={isDarkMode}
                                    setIsDarkMode={setIsDarkMode}
                                    sites={sites}
                                    activeSiteId={activeSite.id}
                                    inputs={{ ...inputs, baseArea: finalRes.base, floorsAbove: finalRes.floorsAbove }}
                                    res={finalRes}
                                    siteResults={siteResults}
                                    onUpdateBaseDim={updateBaseDim}
                                    onUpdateRoads={handleUpdateRoads}
                                    onUpdateInputs={handleUpdateInputs}
                                    onUpdateSiteSetting={handleUpdateSiteSetting}
                                    showCheckLines={showCheckLines}
                                    setShowCheckLines={setShowCheckLines}
                                    showHeightRatio={showHeightRatio}
                                    setShowHeightRatio={setShowHeightRatio}
                                    showWeiLao={showWeiLao}
                                    setShowWeiLao={setShowWeiLao}
                                    showBackyard={showBackyard}
                                    setShowBackyard={setShowBackyard}
                                    isConstraintMode={isConstraintMode}
                                    setIsConstraintMode={setIsConstraintMode}
                                    onUnusedAreaChange={setUnusedArea}
                                    placementMode={placementMode}
                                    referenceEdge={referenceEdge}
                                    setReferenceEdge={setReferenceEdge}
                                    placementOffset={placementOffset}
                                    setPlacementOffset={setPlacementOffset}
                                    onConfirmPlacement={handleConfirmPlacement}
                                    onUpdateSiteGeometry={(geom) => handleUpdateSiteGeometry(activeSiteIndex, geom)}
                                    setActiveSiteIndex={setActiveSiteIndex}
                                />
                            </div>
                        )}
                    </div>
                </section>

                {/* INFO & PLANNING SECTION */}
                <section className="space-y-4">
                    <div className="theme-panel bg-[var(--text-primary)] p-0">
                        <div 
                            className="p-6 flex justify-between items-center cursor-pointer hover:bg-[var(--text-accent)] transition-colors group" 
                            onClick={() => setShowBasePanels(!showBasePanels)}
                        >
                            <h3 className="text-2xl font-black text-white flex items-center gap-3">
                                <Layers size={24} strokeWidth={3} /> 基本資訊與規劃
                            </h3>
                            <div className="text-white group-hover:scale-125 transition-transform">
                                {showBasePanels ? <ChevronUp size={28} strokeWidth={3} /> : <ChevronDown size={28} strokeWidth={3} />}
                            </div>
                        </div>
                    </div>

                    <div className={`transition-all duration-500 ease-in-out ${showBasePanels ? 'opacity-100' : 'opacity-0 h-0 overflow-hidden'}`}>
                        <div className="grid grid-cols-1 xl:grid-cols-3 gap-8">
                            <CollapsibleSection title="基本資料" icon={<MapPin size={20} className="text-[var(--text-accent)]" />} className="xl:col-span-1 theme-panel p-6 bg-white">
                                <div className="space-y-6">
                                    <div>
                                        <label className={labelClass}>行政區</label>
                                        <input type="text" name="district" value={inputs.district} onChange={handleInputChange} className={inputClass} />
                                    </div>
                                    <div className="space-y-4">
                                        <div className="flex justify-between items-center border-b-4 border-[var(--border-strong)] pb-2">
                                            <label className={labelClass}>地號與面積 (㎡)</label>
                                            <button onClick={addLot} className="font-black text-[var(--text-accent)] hover:scale-110 transition-transform flex items-center gap-1">
                                                <Plus size={16} strokeWidth={3} /> 新增
                                            </button>
                                        </div>
                                        <div className="max-h-60 overflow-y-auto pr-2 space-y-3 thin-scrollbar">
                                            {lots.map(l => (
                                                <div key={l.id} className="flex gap-2 items-center">
                                                    <input type="text" placeholder="地號" value={l.lot} onChange={(e) => updateLot(l.id, 'lot', e.target.value)} className={`${inputClass} text-xs`} />
                                                    <input type="number" placeholder="面積" value={l.area} onChange={(e) => updateLot(l.id, 'area', e.target.value)} className={`${inputClass} text-xs w-28 text-right`} />
                                                    {lots.length > 1 && (
                                                        <button onClick={() => removeLot(l.id)} className="p-2 border-2 border-[var(--border-strong)] hover:bg-red-500 hover:text-white transition-colors">
                                                            <Trash2 size={16} />
                                                        </button>
                                                    )}
                                                </div>
                                            ))}
                                        </div>
                                        <div className="text-right text-lg font-black bg-[var(--text-primary)] text-white p-2">
                                            合計: {formatNum(totalBaseArea)} ㎡ ({formatNum(toPing(totalBaseArea))} 坪)
                                        </div>
                                    </div>
                                    
                                    <div className="space-y-4 pt-6 mt-6 border-t-8 border-[var(--border-strong)]">
                                        <div className="flex justify-between items-center">
                                            <label className={labelClass}>臨路狀況</label>
                                            <button onClick={addRoad} className="font-black text-[var(--text-accent)] hover:scale-110 transition-transform flex items-center gap-1">
                                                <Plus size={16} strokeWidth={3} /> 新增道路
                                            </button>
                                        </div>
                                        <div className="space-y-3">
                                            {inputs.roads.map(r => (
                                                <div key={r.id} className="theme-panel p-4 bg-stone-100 border-2">
                                                    <div className="grid grid-cols-2 gap-3 mb-3">
                                                        <div>
                                                            <label className="text-[10px] font-black uppercase">路名</label>
                                                            <input type="text" value={r.name} onChange={(e) => updateRoad(r.id, 'name', e.target.value)} className={`${inputClass} py-1 text-xs`} />
                                                        </div>
                                                        <div>
                                                            <label className="text-[10px] font-black uppercase text-center block">寬(M)</label>
                                                            <input type="number" value={r.width} onChange={(e) => updateRoad(r.id, 'width', e.target.value)} className={`${inputClass} py-1 text-xs text-center`} />
                                                        </div>
                                                    </div>
                                                    <div className="flex justify-between items-center">
                                                        <div className="text-[10px] font-black uppercase p-1 bg-white border border-black">
                                                            方位: {r.edgeIndex !== undefined ? `邊線 ${r.edgeIndex}` : r.orientation}
                                                        </div>
                                                        <div className="flex items-center gap-4">
                                                            <label className="flex items-center gap-2 text-xs font-black cursor-pointer">
                                                                <input type="checkbox" checked={r.type === 'primary'} onChange={(e) => updateRoad(r.id, 'type', e.target.checked ? 'primary' : 'secondary')} className="w-5 h-5 border-4 border-black checked:bg-black rounded-none appearance-none checked:after:content-['✓'] after:text-white after:flex after:justify-center after:items-center" />
                                                                主要道路
                                                            </label>
                                                            {inputs.roads.length > 1 && (
                                                                <button onClick={() => removeRoad(r.id)} className="text-red-500 hover:scale-125 transition-transform"><Trash2 size={18} /></button>
                                                            )}
                                                        </div>
                                                    </div>
                                                </div>
                                            ))}
                                        </div>
                                    </div>
                                </div>
                            </CollapsibleSection>

                            <CollapsibleSection title="法規參數" icon={<FileText size={20} className="text-sky-500" />} className="xl:col-span-1 theme-panel p-6 bg-white">
                                <div className="space-y-6">
                                    <div><label className={labelClass}>使用分區</label><input type="text" name="zoneType" value={inputs.zoneType} onChange={handleInputChange} className={inputClass} /></div>
                                    <div><label className={labelClass}>都市計畫名稱</label><input type="text" name="urbanPlanName" value={inputs.urbanPlanName} onChange={handleInputChange} className={inputClass} /></div>
                                    <div className="grid grid-cols-2 gap-4">
                                        <div className="bg-sky-50 p-3 border-2 border-sky-500">
                                            <label className={`${labelClass} text-sky-700`}>建蔽率 (%)</label>
